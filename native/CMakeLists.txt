cmake_minimum_required(VERSION 3.24)
project(live2d_native_bridge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH "$ORIGIN;@loader_path")
set(PURISM_CORE_SOURCE_DIR "" CACHE PATH "Optional local Purism Core checkout")
include(FetchContent)
if(NOT PURISM_CORE_SOURCE_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/purism-core/CMakeLists.txt")
  set(PURISM_CORE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/purism-core")
endif()
set(PURISM_CORE_VERSION_RC OFF CACHE BOOL "" FORCE)
if(PURISM_CORE_SOURCE_DIR)
  FetchContent_Declare(purism_core SOURCE_DIR "${PURISM_CORE_SOURCE_DIR}")
else()
  FetchContent_Declare(purism_core
    GIT_REPOSITORY https://github.com/SakuraMotion/PurismCore.git
    GIT_TAG 1069334965522df5d0b791e97f01e26b19c45456)
endif()
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz)
FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.91.9b.tar.gz)
FetchContent_Declare(json URL https://github.com/nlohmann/json/archive/refs/tags/v3.11.3.tar.gz)
FetchContent_Declare(miniaudio URL https://github.com/mackron/miniaudio/archive/refs/tags/0.11.22.tar.gz)
FetchContent_Declare(stb URL https://github.com/nothings/stb/archive/f0569113c93ad095470c54bf34a17b36646bbbb5.tar.gz)
FetchContent_MakeAvailable(glfw imgui json miniaudio stb purism_core)
find_package(OpenGL REQUIRED)
add_library(studio_ui STATIC
  "${imgui_SOURCE_DIR}/imgui.cpp" "${imgui_SOURCE_DIR}/imgui_draw.cpp"
  "${imgui_SOURCE_DIR}/imgui_tables.cpp" "${imgui_SOURCE_DIR}/imgui_widgets.cpp"
  "${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp" "${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp")
target_include_directories(studio_ui PUBLIC "${imgui_SOURCE_DIR}" "${imgui_SOURCE_DIR}/backends")
target_link_libraries(studio_ui PUBLIC glfw OpenGL::GL)
add_library(live2d_native_bridge SHARED bridge.cpp ui.cpp audio.c)
set(L2D_BRIDGE_OUTPUT_NAME "live2d_native_bridge" CACHE STRING "Bridge filename for side-by-side update testing")
set_target_properties(live2d_native_bridge PROPERTIES OUTPUT_NAME "${L2D_BRIDGE_OUTPUT_NAME}")
target_include_directories(live2d_native_bridge PRIVATE "${glfw_SOURCE_DIR}/deps" "${stb_SOURCE_DIR}" "${miniaudio_SOURCE_DIR}")
target_link_libraries(live2d_native_bridge PRIVATE studio_ui nlohmann_json::nlohmann_json OpenGL::GL PurismCore::PurismCore)
target_compile_definitions(live2d_native_bridge PRIVATE GLFW_INCLUDE_NONE)
if(UNIX AND NOT APPLE)
  target_link_libraries(live2d_native_bridge PRIVATE dl pthread m)
endif()
add_custom_command(TARGET live2d_native_bridge POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "$<TARGET_FILE:PurismCore>" "$<TARGET_FILE_DIR:live2d_native_bridge>")
if(MINGW)
  target_link_options(live2d_native_bridge PRIVATE -static-libgcc -static-libstdc++)
endif()
install(TARGETS live2d_native_bridge glfw PurismCore RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../licenses/PurismCore-LICENSE.txt" DESTINATION licenses)
